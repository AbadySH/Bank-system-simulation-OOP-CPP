cmake_minimum_required(VERSION 3.10)
project(BankSystemSimulation CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
find_package(Threads REQUIRED)

add_executable(bank-server src/main.cpp)
target_link_libraries(bank-server Threads::Threads)
if(WIN32)
  target_link_libraries(bank-server ws2_32)
endif()

add_executable(core_tests tests/core_tests.cpp)
target_link_libraries(core_tests Threads::Threads)
enable_testing()
add_test(NAME core_tests COMMAND core_tests)
